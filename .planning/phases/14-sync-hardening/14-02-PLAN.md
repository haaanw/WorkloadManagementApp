**SyncStatusView.swift** -- New file in WorkloadApp/Views/Profile/. Per UI-SPEC:

```swift
import SwiftUI

struct SyncStatusView: View {
    private let store = SyncTimestampStore.shared
    @Environment(AppContainer.self) private var container
    @Environment(\.modelContext) private var modelContext

    var body: some View {
        ScrollView {
            VStack(spacing: 0) {
                // Section header per UI-SPEC: "SYNC STATUS" in 13pt Alpino Regular, ColorTokens.text2, all-caps, +0.08em tracking
                Text("SYNC STATUS")
                    .font(.Tokens.micro)
                    .foregroundStyle(ColorTokens.text2)
                    .tracking(0.88)
                    .frame(maxWidth: .infinity, alignment: .leading)
                    .padding(.horizontal, 16)
                    .padding(.top, 24)
                    .padding(.bottom, 8)

                // Card container per UI-SPEC: Rectangle with divider border, 0pt corners
                VStack(spacing: 0) {
                    ForEach(Array(SyncEntity.allCases.enumerated()), id: \.element.id) { index, entity in
                        entityRow(entity)
                        if index < SyncEntity.allCases.count - 1 {
                            Rectangle()
                                .fill(ColorTokens.divider)
                                .frame(height: 0.5)
                                .padding(.leading, 40)
                        }
                    }
                }
                .padding(.horizontal, 16)
            }
        }
        .background(ColorTokens.background)
        .navigationTitle("Sync Status")
        .navigationBarTitleDisplayMode(.inline)
        .refreshable {
            // Pull-to-refresh triggers immediate sync per UI-SPEC interaction contract
            guard !store.isSyncing else { return }
            await container.syncService.pushAll(context: modelContext)
            await container.syncService.pullAll(context: modelContext)
        }
    }

    @ViewBuilder
    private func entityRow(_ entity: SyncEntity) -> some View {
        let hasFailed = store.lastErrors[entity] != nil
        let lastSync = store.lastSuccess(for: entity)

        HStack(spacing: 8) {
            // Leading status indicator: 8pt circle per UI-SPEC
            Circle()
                .fill(hasFailed ? ColorTokens.zoneCaution : ColorTokens.zoneOptimal)
                .frame(width: 8, height: 8)

            // Center VStack: entity name + timestamp
            VStack(alignment: .leading, spacing: 2) {
                Text(entity.displayName)
                    .font(.Tokens.label)
                    .foregroundStyle(ColorTokens.text1)

                if hasFailed, let error = store.lastErrors[entity] {
                    Text("Failed \(error.timestamp, formatter: Self.relativeFormatter)")
                        .font(.Tokens.micro)
                        .foregroundStyle(ColorTokens.text2)
                } else if let date = lastSync {
                    Text(date, formatter: Self.relativeFormatter)
                        .font(.Tokens.micro)
                        .foregroundStyle(ColorTokens.text2)
                } else {
                    Text("Never")
                        .font(.Tokens.micro)
                        .foregroundStyle(ColorTokens.text2)
                }
            }

            Spacer()

            // Trailing error text (only when failed) per UI-SPEC
            if let error = store.lastErrors[entity] {
                Text(error.message)
                    .font(.Tokens.micro)
                    .foregroundStyle(ColorTokens.zoneCaution)
                    .lineLimit(1)
            }
        }
        .padding(.horizontal, 16)
        .frame(minHeight: 48)
        .accessibilityElement(children: .combine)
        .accessibilityLabel(hasFailed
            ? "\(entity.displayName), sync failed, \(store.lastErrors[entity]?.message ?? "")"
            : "\(entity.displayName), synced \(lastSync.map { Self.relativeFormatter.string(from: $0) } ?? "never")")
    }

    private static let relativeFormatter: RelativeDateTimeFormatter = {
        let f = RelativeDateTimeFormatter()
        f.unitsStyle = .short
        return f
    }()
}
```

Note: `RelativeDateTimeFormatter` is a Foundation class that produces "2 min. ago", "3 hr. ago" etc. Use `.string(from:)` or use the SwiftUI `Text(date, formatter:)` approach. The SwiftUI `Text` with formatter auto-updates. If `Text(date, formatter:)` doesn't work with `RelativeDateTimeFormatter`, use `Text(relativeFormatter.localizedString(for: date, relativeTo: Date()))` instead.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| SyncTimestampStore -> UI | Observable state drives badge visibility; incorrect state = misleading UI |
| User pull-to-refresh -> SyncService | User-triggered sync could race with background sync |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-05 | Information Disclosure | SyncStatusView shows entity names and error types | accept | Entity names and generic error categories ("Network unavailable") reveal no sensitive data. No PII, no API keys, no user data in error messages. |
| T-14-06 | Denial of Service | Pull-to-refresh spam could trigger excessive Supabase calls | mitigate | isSyncing guard in SyncTimestampStore prevents concurrent sync cycles. SwiftUI .refreshable has built-in rate limiting. |
| T-14-07 | Tampering | clearAll() could be called at wrong time, losing valid timestamps | accept | clearAll() is only called from signOut/deleteAccount -- both are intentional user actions that destroy all local data anyway. |
</threat_model>
