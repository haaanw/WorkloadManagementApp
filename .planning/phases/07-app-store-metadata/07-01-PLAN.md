                            // Google "G" logo placeholder -- use Text "G" styled

                                .foregroundStyle(ColorTokens.text1)

```swift
private func handleAppleSignIn(credential: ASAuthorizationAppleIDCredential) async {
    isSocialLoading = true
    errorMessage = nil
    do {
        try await container.authService.signInWithApple(credential: credential)
        // Same bootstrap as email sign-in
        let localAthletes = try? modelContext.fetch(FetchDescriptor<Athlete>())
        if localAthletes?.isEmpty != false {
            guard let userId = await container.authService.currentUserId() else {
                throw AuthBootstrapError.noUserId
            }
            let athlete = await container.syncService.bootstrapAthlete(
                context: modelContext, userId: userId
            )
            if athlete == nil {
                throw AuthBootstrapError.athleteNotFound
            }
        }
        await container.syncService.pullAll(context: modelContext)
        isSocialLoading = false
        container.setAuthenticated(true)
    } catch {
        errorMessage = error.localizedDescription
        isSocialLoading = false
    }
}

private func handleGoogleSignIn() async {
    isSocialLoading = true
    errorMessage = nil
    do {
        try await container.authService.signInWithGoogle()
        let localAthletes = try? modelContext.fetch(FetchDescriptor<Athlete>())
        if localAthletes?.isEmpty != false {
            guard let userId = await container.authService.currentUserId() else {
                throw AuthBootstrapError.noUserId
            }
            let athlete = await container.syncService.bootstrapAthlete(
                context: modelContext, userId: userId
            )
            if athlete == nil {
                throw AuthBootstrapError.athleteNotFound
            }
        }
        await container.syncService.pullAll(context: modelContext)
        isSocialLoading = false
        container.setAuthenticated(true)
    } catch {
        errorMessage = error.localizedDescription
        isSocialLoading = false
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client -> Supabase Auth | Social auth credentials (Apple ID token, Google OAuth code) cross from device to backend |
| Google OAuth redirect -> App | OAuth callback URL carries session token from browser back to app |
| Apple credential -> AuthService | Apple identity token is single-use, must not be cached or logged |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-01 | Spoofing | Apple Sign-In | mitigate | Supabase validates Apple ID token server-side (nonce, signature, issuer) -- never trust client-side credential alone |
| T-07-02 | Spoofing | Google Sign-In | mitigate | Supabase handles PKCE and state validation in OAuth flow -- no custom token parsing |
| T-07-03 | Information Disclosure | AuthService | mitigate | Never log or cache Apple identity tokens -- they are single-use for signInWithIdToken |
| T-07-04 | Elevation of Privilege | OAuth callback | mitigate | URL scheme `com.tonus.app://` registered in Info.plist; Supabase validates callback origin and state parameter |
| T-07-05 | Spoofing | Account linking | accept | Supabase auto-links accounts with same verified email -- risk of unintended linking is low since email verification is required. User assumption A1 needs dashboard verification. |
| T-07-06 | Denial of Service | Social login buttons | accept | Rate limiting handled by Apple/Google auth servers, not app-side concern |
</threat_model>
