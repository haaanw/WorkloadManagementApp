```swift
import Foundation
import SwiftData

@MainActor
final class BehaviorTagRepository {
    private let modelContext: ModelContext

    init(modelContext: ModelContext) {
        self.modelContext = modelContext
    }

    func fetchTagsForDate(_ date: Date) throws -> [BehaviorTag] {
        let start = Calendar.current.startOfDay(for: date)
        let end = Calendar.current.date(byAdding: .day, value: 1, to: start)!
        let predicate = #Predicate<BehaviorTag> { $0.date >= start && $0.date < end }
        let descriptor = FetchDescriptor<BehaviorTag>(predicate: predicate, sortBy: [SortDescriptor(\.tagName)])
        return try modelContext.fetch(descriptor)
    }

    func fetchAllTags(days: Int = 90) throws -> [BehaviorTag] {
        let start = Calendar.current.date(byAdding: .day, value: -days, to: .now)!
        let predicate = #Predicate<BehaviorTag> { $0.date >= start }
        let descriptor = FetchDescriptor<BehaviorTag>(predicate: predicate, sortBy: [SortDescriptor(\.date)])
        return try modelContext.fetch(descriptor)
    }

    func fetchActiveTags(days: Int = 90) throws -> [BehaviorTag] {
        let start = Calendar.current.date(byAdding: .day, value: -days, to: .now)!
        let predicate = #Predicate<BehaviorTag> { $0.date >= start && $0.isActive }
        let descriptor = FetchDescriptor<BehaviorTag>(predicate: predicate, sortBy: [SortDescriptor(\.date)])
        return try modelContext.fetch(descriptor)
    }

    func fetchCustomTagNames(for athlete: Athlete) throws -> [String] {
        let athleteId = athlete.id
        let predicate = #Predicate<BehaviorTag> { $0.isCustom && $0.athlete?.id == athleteId }
        let descriptor = FetchDescriptor<BehaviorTag>(predicate: predicate)
        let tags = try modelContext.fetch(descriptor)
        return Array(Set(tags.map { $0.tagName })).sorted()
    }

    func upsertTag(_ tag: BehaviorTag) {
        // T-03-01 mitigation: sanitize tag name before persisting
        tag.tagName = String(tag.tagName.trimmingCharacters(in: .whitespaces).prefix(20))
        tag.updatedAt = .now
        modelContext.insert(tag)
        try? modelContext.save()
    }

    func deleteCustomTag(named tagName: String, for athlete: Athlete) throws {
        let athleteId = athlete.id
        let predicate = #Predicate<BehaviorTag> { $0.tagName == tagName && $0.isCustom && $0.athlete?.id == athleteId }
        let descriptor = FetchDescriptor<BehaviorTag>(predicate: predicate)
        let tags = try modelContext.fetch(descriptor)
        for tag in tags {
            modelContext.delete(tag)
        }
        try modelContext.save()
    }
}
```

```sql
CREATE TABLE IF NOT EXISTS behavior_tags (

    athlete_id UUID NOT NULL REFERENCES athletes(id) ON DELETE CASCADE,
    date TIMESTAMPTZ NOT NULL,

-- Enable RLS
ALTER TABLE behavior_tags ENABLE ROW LEVEL SECURITY;

-- RLS policy: athletes can only access their own tags
CREATE POLICY "athlete_own_tags" ON behavior_tags

    USING (athlete_id = auth.uid()::uuid)

CREATE INDEX idx_behavior_tags_athlete_date ON behavior_tags(athlete_id, date);
```

**Create `WorkloadApp/Services/PeriodizationEngine.swift`** -- pure struct, static methods, following AnalyticsEngine pattern per D-01, INTEL-01, INTEL-03:

```swift
import Foundation

/// Detects the athlete's current training phase from load history.
/// Pure struct with static methods -- no state, no side effects.
struct PeriodizationEngine {

    enum TrainingPhase: String, Codable {
        case building    // Volume increasing, intensity stable/moderate
        case pushing     // Intensity increasing, volume stable/decreasing
        case tapering    // Both volume and intensity decreasing
        case maintaining // Stable load, no clear trend
    }

    struct PhaseResult {
        let phase: TrainingPhase
        let confidence: Double    // 0.0 to 1.0
        let weeksSinceTransition: Int

        var displayLabel: String {
            switch phase {
            case .building:    return "Building"
            case .pushing:     return "Pushing"
            case .tapering:    return "Tapering"
            case .maintaining: return "Maintaining"
            }
        }
    }

    struct SufficiencyResult {
        let isSufficient: Bool
        let weeksAvailable: Int
        let weeksRequired: Int         // 8
        let avgSessionsPerWeek: Double
    }
}
```

`checkSufficiency(sessions:minimumWeeks:minimumSessionsPerWeek:)` -- count weeks from earliest session to now, compute avg sessions/week. Returns SufficiencyResult. Minimum 8 weeks and 3.0 sessions/week per INTEL-03.

`detectPhase(workloadSnapshots:sessions:windowWeeks:)` -- algorithm:
1. Group sessions into weekly buckets (last 6 weeks minimum).
2. Compute weekly total volume and weekly average RPE for each week.
3. Split into "recent" window (last 3 weeks) and "previous" window (3 weeks before that).
4. Compute mean weekly volume and mean weekly RPE for each window.
5. Calculate volume change % = (recent - previous) / previous * 100.
6. Calculate intensity change % = (recent RPE - previous RPE) / previous RPE * 100.
7. Classify:
   - **Building**: volume change > +10% AND intensity change < +10%
   - **Pushing**: intensity change > +10% AND volume change < +10%
   - **Tapering**: volume change < -10% AND intensity change < -10%
   - **Maintaining**: none of the above thresholds met
8. Confidence = magnitude of the dominant signal (clamped 0.5 to 1.0).
9. Guard: if recent window has < 2 sessions/week average, return nil (per Pitfall 1 -- rest weeks should not classify).
10. Return PhaseResult with weeksSinceTransition estimated from when the trend direction changed.

```swift
import Foundation

/// Identifies recurring recovery dips correlated with training load spikes.
/// Pure struct with static methods -- no state, no side effects.
struct FatiguePatternEngine {

    struct Insight {
        let text: String          // Natural language: "Recovery typically drops 8 points 2 days after high-volume sessions"
        let confidence: Double    // 0.0 to 1.0
        let sampleSize: Int
    }
}
```

`detectPatterns(workloadSnapshots:recoverySnapshots:sessions:minimumSamples:)` -- algorithm:
1. Build a date-indexed dictionary of daily TSS (from workloadSnapshots, using the `atl` field as proxy, or compute from sessions if `tss` available) and daily recovery score.
2. Compute rolling mean TSS over last 14 days.
3. Identify "high load" days: TSS > 1.5x rolling 14-day mean.
4. For each lag in [1, 2, 3] days:
   a. Collect recovery score deltas (score on lag day minus score on high-load day) for all high-load days.
   b. Collect recovery score deltas for non-high-load days at the same lag.
   c. Compute mean delta for high-load group and mean delta for non-high-load group.
   d. If high-load mean delta is worse by > 5 points AND sample size >= `minimumSamples` (default 5):
      - Generate insight: "Recovery typically drops {abs(delta)} points {lag} days after high-volume sessions"
5. Also check by exercise category (if WorkoutSession has dominant category info):
   - Group high-load days by the primary exercise category of that session.
   - If a specific category shows a stronger pattern, generate: "Recovery typically drops {N} points {lag} days after high-volume {category} sessions"
6. Return insights sorted by confidence descending, capped at 5.
7. Confidence = min(1.0, sampleSize / 15.0) * (abs(delta_difference) / 20.0) clamped to 0.3-1.0.

```swift
import Foundation

/// Computes recovery impact percentages for behavior tags.
/// Pure struct with static methods -- no state, no side effects.
struct BehaviorCorrelationEngine {

    struct TagCorrelation {
        let tagName: String
        let recoveryWithTag: Double      // mean recovery score on days with tag active
        let recoveryWithoutTag: Double   // mean recovery score on days without tag active
        let impactPercentage: Double     // (with - without) / without * 100
        let sampleCountWith: Int
        let sampleCountWithout: Int
        let isSufficient: Bool           // both counts >= minimumSamplesPerGroup
    }

    struct SufficiencyInfo {
        let tagName: String
        let daysWithTag: Int
        let daysWithoutTag: Int
        let neededWith: Int     // max(0, 5 - daysWithTag)
        let neededWithout: Int  // max(0, 5 - daysWithoutTag)
    }
}
```

`computeCorrelations(tags:recoverySnapshots:minimumSamplesPerGroup:)`:
1. Group tags by tagName.
2. For each unique tag name, partition dates into "tag active" (isActive == true on that date) and "tag not active" (either isActive == false or no tag record for that date within the analysis window).
3. Match each date to the closest recovery score (same day or next morning).
4. Compute mean recovery for with-tag and without-tag groups.
5. Only return TagCorrelation where BOTH groups have >= `minimumSamplesPerGroup` (default 5) per D-09.
6. impactPercentage = (recoveryWithTag - recoveryWithoutTag) / recoveryWithoutTag * 100, rounded to 1 decimal.

`checkSufficiency(tags:minimumSamplesPerGroup:)`:
1. For each unique tag name, count days with tag active and days without.
2. Return SufficiencyInfo array showing how many more days needed per tag.
