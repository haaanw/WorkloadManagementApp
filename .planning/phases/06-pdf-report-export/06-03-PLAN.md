```swift
var body: some View {
    NavigationStack {
        VStack(spacing: 0) {
            // Title section
            VStack(alignment: .leading, spacing: 8) {
                Text("Select Athletes")
                    .font(.Tokens.sectionHead)
                    .foregroundStyle(ColorTokens.text1)
                Text("Choose athletes to include in the roster report")
                    .font(.Tokens.label)
                    .foregroundStyle(ColorTokens.text2)
            }
            .frame(maxWidth: .infinity, alignment: .leading)
            .padding(24)

            Rectangle().fill(ColorTokens.divider).frame(height: 0.5)

            // Athlete list with checkmarks
            ScrollView {
                VStack(spacing: 0) {
                    ForEach(viewModel.linkedAthletes, id: \.id) { athlete in
                        athleteRow(athlete)
                        Rectangle().fill(ColorTokens.divider).frame(height: 0.5)
                    }
                }
            }

            Rectangle().fill(ColorTokens.divider).frame(height: 0.5)

            // Date range chips (same pattern as PDFGenerationSheet)
            dateRangeChips
                .padding(.horizontal, 24)
                .padding(.vertical, 16)

            // CTA button
            generateButton
                .padding(.horizontal, 24)
                .padding(.bottom, 16)
        }
        .background(ColorTokens.background)
        .toolbar {
            ToolbarItem(placement: .cancellationAction) {
                Button("Cancel") { dismiss() }
                    .font(.Tokens.label)
                    .foregroundStyle(ColorTokens.text1)
            }
            ToolbarItem(placement: .primaryAction) {
                Button(allSelected ? "Deselect All" : "Select All") {
                    toggleSelectAll()
                }
                .font(.Tokens.label)
                .foregroundStyle(ColorTokens.text1)
            }
        }
    }
    .onAppear {
        if !didInitialize {
            selectedAthleteIds = Set(viewModel.linkedAthletes.map(\.id))
            didInitialize = true
        }
    }
    .sheet(isPresented: $showShareSheet) {
        if let url = exportFileURL { ShareSheet(items: [url]) }
    }
    .alert("Error", isPresented: .constant(errorMessage != nil)) {
        Button("OK") { errorMessage = nil }
    } message: { Text(errorMessage ?? "") }
}
```

**generateButton:**
- Same styling as PDFGenerationSheet CTA
- Label: "Generate Roster Report" / "Generating..."
- Disabled when `isGenerating` or `selectedAthleteIds.isEmpty`

**generateReport() method:**
1. Set `isGenerating = true`
2. Build `[RosterAthleteData]` from selected athletes:
   ```swift
   let selectedAthletes = viewModel.linkedAthletes.filter { selectedAthleteIds.contains($0.id) }
   let rosterData: [RosterAthleteData] = selectedAthletes.map { athlete in
       let workload = viewModel.latestWorkloadSnapshot[athlete.id]
       let recovery = viewModel.latestRecoverySnapshot[athlete.id]
       let sessions = (try? athlete.sessions
           .filter { $0.sessionDate >= Calendar.current.date(byAdding: .day, value: -selectedRange.days, to: .now)! }) ?? []
       let streak = StreakEngine.computeStreak(sessions: Array(athlete.sessions))
       return RosterAthleteData(
           name: athlete.displayName,
           recoveryScore: recovery?.recoveryScore,
           acwrZone: workload?.zone ?? .noData,
           sessionsCount: sessions.count,
           streakCount: streak,
           isOverreaching: workload?.zone == .danger
       )
   }
   ```
   Note: Access `athlete.sessions` via the relationship array (WorkoutSession relationship on Athlete model). If sessions relationship is named differently, check the Athlete model.
3. Get coach name: `athletes.first?.displayName ?? "Coach"`
4. Call `PDFReportEngine.generateCoachRosterReport(coachName:athletes:dateRange:)`
5. Write Data to temp file, set `exportFileURL`, `showShareSheet = true`
6. On error: set `errorMessage = "Report generation failed. Please try again."`
7. Set `isGenerating = false`

**allSelected computed property:** `selectedAthleteIds.count == viewModel.linkedAthletes.count`

**toggleSelectAll():** If all selected, clear set. Otherwise, add all linked athlete IDs.
